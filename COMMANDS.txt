🔧 DOCKER COMMANDS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
docker-compose up -d          # Start all services
docker-compose down           # Stop all services
docker-compose restart        # Restart all services
docker-compose ps             # Show running containers
docker-compose logs           # View all logs
docker-compose logs -f        # Follow logs (real-time)
docker-compose logs backend   # View backend logs only

🔍 DEBUGGING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
docker-compose logs backend   # Backend issues
docker-compose logs postgres  # Database issues
docker-compose logs frontend  # Frontend issues
docker ps                     # List all containers
docker exec -it smartbank_backend bash   # Enter backend container
docker exec -it smartbank_db psql -U postgres -d smartbank  # Enter database

🗄️ DATABASE COMMANDS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
# Inside database container:
\dt                          # List all tables
\d users                     # Describe users table
SELECT * FROM users;         # View all users
SELECT * FROM accounts;      # View all accounts
SELECT * FROM transactions;  # View all transactions

# From outside:
docker exec smartbank_db psql -U postgres -d smartbank -c "SELECT * FROM users;"

🔄 RESET & CLEAN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
docker-compose restart        # Soft restart
docker-compose down -v        # Remove everything (including data!)
docker-compose up -d --build  # Rebuild and start
./quick-start.sh              # Complete fresh start
